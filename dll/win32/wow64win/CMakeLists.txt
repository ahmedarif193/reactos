spec2def(wow64win.dll wow64win.spec ADD_IMPORTLIB)

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    ${REACTOS_SOURCE_DIR}/win32ss/w32ksvc32.h
    ${REACTOS_SOURCE_DIR}/win32ss/win32u/win32u.spec
    ${CMAKE_CURRENT_SOURCE_DIR}/gdi.c
    ${CMAKE_CURRENT_SOURCE_DIR}/user.c)

set(_wow64win_wrappers)
foreach(_source gdi.c user.c)
    file(STRINGS ${CMAKE_CURRENT_SOURCE_DIR}/${_source} _lines REGEX "^NTSTATUS WINAPI wow64_Nt(User|Gdi)")
    foreach(_line ${_lines})
        string(REGEX MATCH "wow64_(Nt(User|Gdi)[A-Za-z0-9_]+)" _match "${_line}")
        list(APPEND _wow64win_wrappers ${CMAKE_MATCH_1})
    endforeach()
endforeach()

set(_wow64win_exports)
file(STRINGS ${REACTOS_SOURCE_DIR}/win32ss/win32u/win32u.spec _lines REGEX "^@ .*Nt(User|Gdi)")
foreach(_line ${_lines})
    string(REGEX MATCH " (Nt(User|Gdi)[A-Za-z0-9_]+)\\(" _match "${_line}")
    list(APPEND _wow64win_exports ${CMAKE_MATCH_1})
endforeach()

set(_wow64win_missing_exports)
set(_wow64win_native_stubs "/* Generated for Wine thunks without a ReactOS win32u export. */\n\n")
foreach(_name IN LISTS _wow64win_wrappers)
    list(FIND _wow64win_exports ${_name} _export_index)
    if(_export_index EQUAL -1)
        list(APPEND _wow64win_missing_exports ${_name})
        string(APPEND _wow64win_native_stubs "#define ${_name}(...) 0\n")
    endif()
endforeach()
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/native_stubs.h "${_wow64win_native_stubs}")

set(_wow64win_syscalls_before)
set(_wow64win_syscalls_debug)
set(_wow64win_syscalls_after)
set(_wow64win_stubs_before)
set(_wow64win_stubs_debug)
set(_wow64win_stubs_after)
set(_wow64win_section before)
set(_wow64win_skip FALSE)
set(_wow64win_missing_service_exports)
set(_wow64win_missing_service_wrappers)

file(STRINGS ${REACTOS_SOURCE_DIR}/win32ss/w32ksvc32.h _lines)
foreach(_line ${_lines})
    string(STRIP "${_line}" _line)
    if(_line STREQUAL "#ifdef _WIN64")
        set(_wow64win_skip TRUE)
    elseif(_line MATCHES "^#endif.*_WIN64")
        set(_wow64win_skip FALSE)
    elseif(_line STREQUAL "#if DBG")
        set(_wow64win_section debug)
    elseif(_line MATCHES "^#endif.*DBG")
        set(_wow64win_section after)
    elseif(NOT _wow64win_skip AND _line MATCHES "^SVC_\\(([A-Za-z0-9_]+),[ \\t]*([0-9]+)\\)")
        set(_name Nt${CMAKE_MATCH_1})
        math(EXPR _bytes "${CMAKE_MATCH_2} * 4")
        list(FIND _wow64win_wrappers ${_name} _wrapper_index)
        list(FIND _wow64win_exports ${_name} _export_index)
        if(_wrapper_index EQUAL -1)
            list(APPEND _wow64win_missing_service_wrappers ${_name})
        elseif(_export_index EQUAL -1)
            list(APPEND _wow64win_missing_service_exports ${_name})
        endif()
        if(_wrapper_index EQUAL -1 OR _export_index EQUAL -1)
            set(_handler wow64_stub_${_name})
            set(_stub "    SYSCALL_STUB( ${_name} ) ")
            string(APPEND _stub "\\")
            string(APPEND _stub "\n")
            string(APPEND _wow64win_stubs_${_wow64win_section} "${_stub}")
        else()
            set(_handler wow64_${_name})
        endif()

        set(_entry "    SYSCALL_ENTRY( 0, ${_name}, ${_handler}, ${_bytes} ) ")
        string(APPEND _entry "\\")
        string(APPEND _entry "\n")
        string(APPEND _wow64win_syscalls_${_wow64win_section} "${_entry}")
    endif()
endforeach()

foreach(_list _wow64win_missing_exports _wow64win_missing_service_exports _wow64win_missing_service_wrappers)
    list(REMOVE_DUPLICATES ${_list})
    list(SORT ${_list})
endforeach()
list(LENGTH _wow64win_missing_exports _wow64win_missing_export_count)
list(LENGTH _wow64win_missing_service_exports _wow64win_missing_service_export_count)
list(LENGTH _wow64win_missing_service_wrappers _wow64win_missing_service_wrapper_count)
string(REPLACE ";" "\n" _wow64win_missing_exports_text "${_wow64win_missing_exports}")
string(REPLACE ";" "\n" _wow64win_missing_service_exports_text "${_wow64win_missing_service_exports}")
string(REPLACE ";" "\n" _wow64win_missing_service_wrappers_text "${_wow64win_missing_service_wrappers}")
set(_wow64win_parity_report "${CMAKE_CURRENT_BINARY_DIR}/win32u-parity.txt")
file(WRITE ${_wow64win_parity_report}
    "Wine wrappers without ReactOS win32u exports (${_wow64win_missing_export_count})\n${_wow64win_missing_exports_text}\n\n"
    "ReactOS services with a Wine wrapper but no win32u export (${_wow64win_missing_service_export_count})\n${_wow64win_missing_service_exports_text}\n\n"
    "ReactOS services without a Wine wrapper (${_wow64win_missing_service_wrapper_count})\n${_wow64win_missing_service_wrappers_text}\n")
message(STATUS "WoW64 win32u parity: ${_wow64win_missing_service_export_count} service export gaps, ${_wow64win_missing_service_wrapper_count} missing Wine wrappers")
add_custom_target(wow64win_parity COMMAND ${CMAKE_COMMAND} -E cat ${_wow64win_parity_report} VERBATIM)

set(_wow64win_header "/* Generated from ReactOS w32ksvc32.h. */\n\n#define WOW64WIN_LIST_END\n\n")
macro(_wow64win_add_fragment _name _variable)
    string(APPEND _wow64win_header "#define ${_name} ")
    string(APPEND _wow64win_header "\\")
    string(APPEND _wow64win_header "\n${${_variable}}    WOW64WIN_LIST_END\n\n")
endmacro()

_wow64win_add_fragment(WOW64WIN_SYSCALLS_BEFORE _wow64win_syscalls_before)
_wow64win_add_fragment(WOW64WIN_SYSCALLS_AFTER _wow64win_syscalls_after)
_wow64win_add_fragment(WOW64WIN_STUBS_BEFORE _wow64win_stubs_before)
_wow64win_add_fragment(WOW64WIN_STUBS_AFTER _wow64win_stubs_after)

string(APPEND _wow64win_header "#if DBG\n")
_wow64win_add_fragment(WOW64WIN_SYSCALLS_DEBUG _wow64win_syscalls_debug)
_wow64win_add_fragment(WOW64WIN_STUBS_DEBUG _wow64win_stubs_debug)
string(APPEND _wow64win_header "#else\n#define WOW64WIN_SYSCALLS_DEBUG WOW64WIN_LIST_END\n#define WOW64WIN_STUBS_DEBUG WOW64WIN_LIST_END\n#endif\n\n")

string(APPEND _wow64win_header "#define ALL_SYSCALLS32 ")
string(APPEND _wow64win_header "\\")
string(APPEND _wow64win_header "\n    WOW64WIN_SYSCALLS_BEFORE ")
string(APPEND _wow64win_header "\\")
string(APPEND _wow64win_header "\n    WOW64WIN_SYSCALLS_DEBUG ")
string(APPEND _wow64win_header "\\")
string(APPEND _wow64win_header "\n    WOW64WIN_SYSCALLS_AFTER\n\n")

string(APPEND _wow64win_header "#define ALL_SYSCALL_STUBS ")
string(APPEND _wow64win_header "\\")
string(APPEND _wow64win_header "\n    WOW64WIN_STUBS_BEFORE ")
string(APPEND _wow64win_header "\\")
string(APPEND _wow64win_header "\n    WOW64WIN_STUBS_DEBUG ")
string(APPEND _wow64win_header "\\")
string(APPEND _wow64win_header "\n    WOW64WIN_STUBS_AFTER\n")

file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/win32syscalls.h "${_wow64win_header}")

add_library(wow64win MODULE
    gdi.c
    syscall.c
    user.c
    ${CMAKE_CURRENT_BINARY_DIR}/wow64win.def)

target_include_directories(wow64win PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
set_module_type(wow64win win32dll ENTRYPOINT DllMain 12)
target_compile_definitions(wow64win PRIVATE __WINESRC__)
add_importlibs(wow64win wow64 win32u ntdll)
add_dependencies(wow64win wow64)
add_cd_file(TARGET wow64win DESTINATION reactos/system32 FOR all)
set_wine_module(wow64win)

    target_compile_options(wow64win PRIVATE -ffunction-sections)
    target_link_options(wow64win PRIVATE -Wl,--gc-sections)
endif()
