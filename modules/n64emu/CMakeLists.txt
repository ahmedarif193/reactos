
set(N64EMU_DIR "" CACHE PATH "Emulator directory deployed to reactos/n64emu in every image")
set(N64EMU_EXE "" CACHE STRING "Emulator executable inside N64EMU_DIR started at logon")
set(N64EMU_ARGS "" CACHE STRING "Extra arguments placed before the ROM path")
set(N64EMU_ROM "" CACHE FILEPATH "N64 ROM deployed to reactos/n64emu/roms and passed to the emulator")

if(N64EMU_DIR AND IS_DIRECTORY "${N64EMU_DIR}")
    file(GLOB_RECURSE _n64emu_files RELATIVE "${N64EMU_DIR}" "${N64EMU_DIR}/*")
    list(FILTER _n64emu_files EXCLUDE REGEX "^Config/(Cheats|Enhancements)/")
    foreach(_file ${_n64emu_files})
        get_filename_component(_subdir "${_file}" DIRECTORY)
        get_filename_component(_name "${_file}" NAME)
        if(_subdir)
            set(_dest "reactos/n64emu/${_subdir}")
        else()
            set(_dest "reactos/n64emu")
        endif()
        add_cd_file(FILE "${N64EMU_DIR}/${_file}" DESTINATION "${_dest}" NAME_ON_CD "${_name}" NO_CAB FOR all)
    endforeach()
    message(STATUS "N64 emulator: deploying ${N64EMU_DIR}")

    if(N64EMU_ROM AND EXISTS "${N64EMU_ROM}")
        get_filename_component(_rom_file "${N64EMU_ROM}" NAME)
        string(REGEX REPLACE "[^A-Za-z0-9._-]" "_" N64EMU_ROM_NAME "${_rom_file}")
        add_cd_file(FILE "${N64EMU_ROM}" DESTINATION reactos/n64emu/roms NAME_ON_CD "${N64EMU_ROM_NAME}" NO_CAB FOR all)
        if(N64EMU_EXE AND EXISTS "${N64EMU_DIR}/${N64EMU_EXE}")
            string(REPLACE "/" "\\" N64EMU_EXE_PATH "${N64EMU_EXE}")
            get_filename_component(_exe_dir "${N64EMU_EXE}" DIRECTORY)
            if(_exe_dir)
                string(REPLACE "/" "\\" N64EMU_EXE_DIR "n64emu/${_exe_dir}")
            else()
                set(N64EMU_EXE_DIR "n64emu")
            endif()
            configure_file(n64emu-launch.cmd.in "${CMAKE_CURRENT_BINARY_DIR}/n64emu-launch.cmd" @ONLY)
            add_cd_file(FILE "${CMAKE_CURRENT_BINARY_DIR}/n64emu-launch.cmd" DESTINATION reactos/n64emu NAME_ON_CD n64emu-launch.cmd NO_CAB FOR all)
            configure_file(hiven64emu.inf.in "${CMAKE_CURRENT_BINARY_DIR}/hiven64emu.inf" @ONLY)
            add_registry_inf("${CMAKE_CURRENT_BINARY_DIR}/hiven64emu.inf")
            message(STATUS "N64 emulator: ${N64EMU_EXE} ${N64EMU_ARGS} autoruns ${N64EMU_ROM_NAME}")
        endif()
    endif()
endif()
