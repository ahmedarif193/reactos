cmake_minimum_required(VERSION 3.24)
project(MesaReactOS LANGUAGES C CXX ASM)
file(STRINGS "${PROJECT_SOURCE_DIR}/VERSION" MESA_VERSION LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/VERSION")

# This is the Windows WGL and Lavapipe build of the vendored Mesa snapshot. It
# deliberately uses its own llvm-mingw toolchain, independent of ReactOS's
# SDK/CRT flags.
if(NOT WIN32 OR NOT MINGW OR NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "Mesa's ReactOS CMake build requires llvm-mingw Clang targeting Windows. See cmake/README.md.")
endif()
set(MESA_ARCH "${CMAKE_SYSTEM_PROCESSOR}" CACHE STRING "Target architecture: arm64, arm64ec, amd64 or i386")
if(MESA_ARCH STREQUAL "aarch64")
    set(MESA_ARCH arm64)
elseif(MESA_ARCH STREQUAL "x86_64")
    set(MESA_ARCH amd64)
elseif(MESA_ARCH MATCHES "^(x86|i686)$")
    set(MESA_ARCH i386)
endif()
if(NOT MESA_ARCH MATCHES "^(arm64|arm64ec|amd64|i386)$")
    message(FATAL_ERROR "Unsupported MESA_ARCH: ${MESA_ARCH}")
endif()
option(MESA_LLVMPIPE "Build LLVMpipe instead of softpipe" OFF)
option(MESA_LAVAPIPE "Build the Lavapipe Vulkan ICD" OFF)
option(MESA_ASSERTIONS "Keep assertions enabled in optimized Mesa builds" ON)
set(MESA_WGL_DLL_NAME "mesa_gallium" CACHE STRING "WGL ICD DLL basename")
if(NOT MESA_WGL_DLL_NAME MATCHES "^[A-Za-z0-9_-]+$")
    message(FATAL_ERROR "MESA_WGL_DLL_NAME must be a DLL basename without an extension.")
endif()
set(MESA_REACTOS_SOURCE_DIR "" CACHE PATH "ReactOS source tree providing KMT and zlib headers")
set(MESA_REACTOS_BUILD_DIR "" CACHE PATH "ReactOS Release build providing KMT and zlib archives")
set(MESA_LLVM_ROOT "" CACHE PATH "Static Windows LLVM installation for the target architecture")
set(MESA_DIRECTX_HEADERS_ROOT "" CACHE PATH "DirectX-Headers installation used by the Windows Vulkan WSI")
set(MESA_PROFILE "${MESA_ARCH}")
set(MESA_V3D FALSE)
if(MESA_ARCH MATCHES "^(arm64|arm64ec|i386)$")
    set(MESA_V3D TRUE)
endif()
if(MESA_LLVMPIPE)
    if(NOT MESA_ARCH MATCHES "^(arm64|arm64ec|amd64)$")
        message(FATAL_ERROR "The ReactOS LLVMpipe configuration supports arm64, arm64ec and amd64 only.")
    endif()
    if(NOT EXISTS "${MESA_LLVM_ROOT}/lib/cmake/llvm/LLVMConfig.cmake")
        message(FATAL_ERROR "MESA_LLVM_ROOT must contain a static Windows LLVM installation for ${MESA_ARCH}.")
    endif()
    set(LLVM_DIR "${MESA_LLVM_ROOT}/lib/cmake/llvm")
    find_package(LLVM REQUIRED CONFIG NO_DEFAULT_PATH)
    if(NOT LLVM_VERSION_MAJOR EQUAL 22 OR LLVM_ENABLE_SHARED_LIBS)
        message(FATAL_ERROR "Mesa LLVMpipe requires static LLVM 22 libraries.")
    endif()
    if(MESA_ARCH MATCHES "^arm64")
        set(_mesa_llvm_target AArch64)
    else()
        set(_mesa_llvm_target X86)
    endif()
    if(NOT _mesa_llvm_target IN_LIST LLVM_TARGETS_TO_BUILD)
        message(FATAL_ERROR "LLVM must include the ${_mesa_llvm_target} target.")
    endif()
    llvm_map_components_to_libnames(MESA_LLVM_LIBRARIES
        bitwriter mcdisassembler mcjit core executionengine scalaropts
        transformutils instcombine coroutines passes ${_mesa_llvm_target})
endif()
if(MESA_V3D)
    foreach(_mesa_header sdk/include/reactos/libs/zlib/zlib.h sdk/include/reactos)
        if(NOT EXISTS "${MESA_REACTOS_SOURCE_DIR}/${_mesa_header}")
            message(FATAL_ERROR "Set MESA_REACTOS_SOURCE_DIR to the ReactOS source tree (missing ${_mesa_header}).")
        endif()
    endforeach()
    set(_mesa_support_archives
        sdk/lib/vc4kmt/libvc4kmt.a
        sdk/lib/3rdparty/zlib/libzlib.a)
    if(MESA_ARCH MATCHES "^arm64")
        list(APPEND _mesa_support_archives
            sdk/lib/rpi3vc4kmt/librpi3vc4kmt.a)
    endif()
    foreach(_mesa_archive IN LISTS _mesa_support_archives)
        if(NOT EXISTS "${MESA_REACTOS_BUILD_DIR}/${_mesa_archive}")
            message(FATAL_ERROR "Build the ReactOS Release KMT/zlib support libraries first (missing ${MESA_REACTOS_BUILD_DIR}/${_mesa_archive}).")
        endif()
    endforeach()
endif()
if(MESA_LAVAPIPE)
    if(NOT MESA_LLVMPIPE)
        message(FATAL_ERROR "MESA_LAVAPIPE requires MESA_LLVMPIPE and its static target LLVM libraries.")
    endif()
    if(NOT EXISTS "${MESA_DIRECTX_HEADERS_ROOT}/include/directx/d3d12.h" OR
       NOT EXISTS "${MESA_DIRECTX_HEADERS_ROOT}/include/dxguids/dxguids.h")
        message(FATAL_ERROR "MESA_DIRECTX_HEADERS_ROOT must contain the installed DirectX-Headers include tree.")
    endif()
    find_program(MESA_GLSLANG_VALIDATOR NAMES glslangValidator REQUIRED)
endif()

find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import mako, packaging, yaml"
    RESULT_VARIABLE _mesa_python_status ERROR_VARIABLE _mesa_python_error)
if(NOT _mesa_python_status EQUAL 0)
    message(FATAL_ERROR "Mesa generators need Python Mako, packaging and PyYAML: ${_mesa_python_error}")
endif()
find_package(BISON 2.7 REQUIRED)
find_package(FLEX REQUIRED)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

include(GNUInstallDirs)
include(cmake/Helpers.cmake)
include(cmake/CompileOptions.cmake)
include(cmake/Generators.cmake)
if(MESA_LAVAPIPE)
    include(cmake/VulkanGenerators.cmake)
endif()
add_custom_target(mesa_generated DEPENDS ${MESA_GENERATED_OUTPUTS})
include(cmake/Targets.cmake)
if(MESA_V3D)
    include(cmake/V3DTargets.cmake)
    include(cmake/D3DTargets.cmake)
endif()
if(MESA_LAVAPIPE)
    include(cmake/VulkanTargets.cmake)
endif()
set_target_properties(mesa_gallium PROPERTIES
    OUTPUT_NAME "${MESA_WGL_DLL_NAME}"
    RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/src/gallium/targets/wgl")
set_target_properties(opengl32 PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/src/gallium/targets/libgl-gdi")
add_custom_target(mesa_wgl ALL DEPENDS mesa_gallium)
install(TARGETS mesa_gallium RUNTIME DESTINATION bin)
if(MESA_V3D)
    add_custom_target(mesa_d3d ALL DEPENDS rpi5vc4d3d)
    install(TARGETS rpi5vc4d3d RUNTIME DESTINATION bin)
endif()
if(MESA_LAVAPIPE)
    add_custom_target(mesa_vulkan ALL DEPENDS vulkan_lvp lvp_icd)
    install(TARGETS vulkan_lvp RUNTIME DESTINATION bin)
    install(FILES "${MESA_LAVAPIPE_MANIFEST}" DESTINATION bin)
endif()
if(MESA_LLVMPIPE)
    set(_mesa_renderer "LLVMpipe")
else()
    set(_mesa_renderer "Softpipe")
endif()
if(MESA_V3D)
    set(_mesa_renderer "V3D + ${_mesa_renderer}")
endif()
message(STATUS "Mesa CMake: ${MESA_ARCH} ${_mesa_renderer}, WGL=${MESA_WGL_DLL_NAME}.dll, Lavapipe=${MESA_LAVAPIPE}")
